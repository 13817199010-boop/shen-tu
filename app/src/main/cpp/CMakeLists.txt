cmake_minimum_required(VERSION 3.22.1)
project(moss_mobile LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(SD_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(SD_WEBP OFF CACHE BOOL "" FORCE)
set(SD_WEBM OFF CACHE BOOL "" FORCE)
set(SD_VULKAN OFF CACHE BOOL "" FORCE)
set(SD_RPC OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_CPU_KLEIDIAI OFF CACHE BOOL "" FORCE)
add_subdirectory(../../../../vendor/stable-diffusion.cpp sdcpp)
add_library(moss_jni SHARED moss_jni.cpp)
target_link_libraries(moss_jni PRIVATE stable-diffusion log)
target_link_options(moss_jni PRIVATE "-Wl,-z,max-page-size=16384")
